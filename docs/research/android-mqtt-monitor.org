** Topic Conventions
   - QoS 0 :: presence / gossip (lossy-ok, TTL-bounded)
   - QoS 1 :: state / claim (at-least-once)
   - QoS 2 :: fault / security events (exactly-once)
   - Retained :: ~agents/{id}/presence~ (last-will on disconnect)

#+begin_src python :tangle src/aq_mqtt_transport.py
"""
aq MQTT transport — Tier 3 (lossy, pub/sub, no ack)
Fits L1.5: presence broadcast, NDJSON payload, TTL via broker retain.
"""
import json, time, os
import paho.mqtt.client as mqtt

BROKER   = os.getenv("AQ_MQTT_BROKER", "localhost")
PORT     = int(os.getenv("AQ_MQTT_PORT", "1883"))
AGENT_ID = os.getenv("AQ_AGENT_ID", "unknown")
BASE     = f"agents/{AGENT_ID}"


def publish_presence(client: mqtt.Client, ttl: int = 30) -> None:
    payload = json.dumps({
        "agent_id": AGENT_ID,
        "ts":       time.time(),
        "ttl":      ttl,
        "host":     os.uname().nodename,
    })
    client.publish(f"{BASE}/presence", payload, qos=0, retain=True)


def publish_gossip(client: mqtt.Client, msg: dict) -> None:
    client.publish(f"{BASE}/gossip", json.dumps(msg), qos=0)


def publish_claim(client: mqtt.Client, repo: str, branch: str, action: str) -> None:
    payload = json.dumps({
        "agent_id": AGENT_ID,
        "repo":     repo,
        "branch":   branch,
        "action":   action,  # announce | withdraw | conflict
        "ts":       time.time(),
    })
    client.publish(f"worktrees/{repo}/{branch}/claim", payload, qos=1)


def connect(on_message=None) -> mqtt.Client:
    c = mqtt.Client(client_id=AGENT_ID, clean_session=False)
    c.tls_set()  # uses system CA; override for self-signed
    lwt = json.dumps({"agent_id": AGENT_ID, "status": "offline"})
    c.will_set(f"{BASE}/presence", lwt, qos=1, retain=True)
    if on_message:
        c.on_message = on_message
    c.connect(BROKER, PORT, keepalive=60)
    return c
#+end_src
